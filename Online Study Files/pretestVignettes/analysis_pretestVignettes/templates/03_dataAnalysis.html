5.1 mean values of valence

library(dplyr)

compute_mean_sd_valence <- function(data,
                                 id_col = "participant_id",
                                 condition_col = "study_condition",
                                 valence_col = "valence") {
  
  data %>%
    mutate(
      valence = as.numeric(.data[[valence_col]])
    ) %>%
    filter(!is.na(valence)) %>%
    group_by(
      participant_id = .data[[id_col]],
      study_condition = .data[[condition_col]]
    ) %>%
    summarise(
      mean_valence = mean(valence, na.rm = TRUE),
      sd_valence   = sd(valence, na.rm = TRUE),
      n_ratings    = n(),
      .groups = "drop"
    )
}

normalize_text <- function(text,
                           lowercase = TRUE,
                           remove_punctuation = TRUE,
                           remove_numbers = TRUE,
                           remove_extra_spaces = TRUE,
                           normalize_umlauts = TRUE) {
  
  # ensure character
  text <- as.character(text)
  
  # convert encoding safely (important for German text)
  text <- iconv(text, to = "UTF-8")
  
  # lowercase
  if (lowercase) {
    text <- tolower(text)
  }
  
  # normalize German umlauts (optional but common)
  if (normalize_umlauts) {
    text <- gsub("ä", "ae", text, fixed = TRUE)
    text <- gsub("ö", "oe", text, fixed = TRUE)
    text <- gsub("ü", "ue", text, fixed = TRUE)
    text <- gsub("ß", "ss", text, fixed = TRUE)
  }
  
  # remove punctuation
  if (remove_punctuation) {
    text <- gsub("[[:punct:]]+", " ", text)
  }
  
  # remove numbers
  if (remove_numbers) {
    text <- gsub("[0-9]+", " ", text)
  }
  
  # remove extra whitespace
  if (remove_extra_spaces) {
    text <- gsub("\\s+", " ", text)
  }
  
  # trim leading/trailing spaces
  text <- trimws(text)
  
  return(text)
}
